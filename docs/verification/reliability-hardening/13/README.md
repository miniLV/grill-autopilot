## Reproduce fixture inputs without launching a model

[seed.json](seed.json) contains the exact ordinary tracked seed files and whole-collection symlink target. It does not launch models, modify the preserved lab or authorize retry. New Git SHAs depend on timestamps/identity and must not be passed off as this run's SHAs. Update the symlink target only to a qualified complete collection when reproducing elsewhere.

```sh
python3 -B - docs/verification/reliability-hardening/13 /absolute/new-smoke13-lab <<'PY'
from pathlib import Path
import json, subprocess, sys
source, lab = map(Path, sys.argv[1:])
data = json.loads((source / "seed.json").read_text())
facts = json.loads((source / "preflight/fixtures.json").read_text())
lab.mkdir()  # Existing paths fail; no reset or cleanup.
seed = lab / "seed"
seed.mkdir()
for name, content in data["files"].items():
    target = seed / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content)
for name, target in data["symlinks"].items():
    link = seed / name
    link.parent.mkdir(parents=True, exist_ok=True)
    link.symlink_to(Path(target).resolve(strict=True), target_is_directory=True)
def run(cwd, *args):
    subprocess.run(args, cwd=cwd, check=True)
def commit(cwd, message):
    run(cwd, "git", "-c", "user.name=Smoke fixture", "-c",
        "user.email=smoke@example.invalid", "commit", "-m", message)
run(seed, "git", "init", "-b", "main")
run(seed, "git", "add", ".")
commit(seed, "Seed legacy baseline")
for name, record in facts["runs"].items():
    checkout = lab / name
    run(lab, "git", "clone", "--no-hardlinks", str(seed), str(checkout))
    run(checkout, "git", "remote", "remove", "origin")
    (checkout / ".grill-autopilot.yaml").write_text(record["config"])
    run(checkout, "git", "add", ".grill-autopilot.yaml")
    commit(checkout, "Configure smoke")
    run(checkout, "python3", "-B", "-m", "unittest", "tests.test_baseline", "-v")
    run(checkout, "python3", "-B", "-m", "unittest", "discover", "-s", "tests", "-v")
PY
```
